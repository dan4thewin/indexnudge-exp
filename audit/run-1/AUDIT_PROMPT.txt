// indexnudge — POC: index a Rust codebase into SQLite for agentic querying
// AI-generated code

use anyhow::{Context, Result};
use clap::Parser;
use rusqlite::Connection;
use std::fs;
use std::path::PathBuf;
use syn::visit::Visit;
use walkdir::WalkDir;

#[derive(Parser)]
#[command(name = "indexnudge", about = "Index a Rust codebase into SQLite")]
struct Cli {
    /// Directory to index
    #[arg(default_value = ".")]
    dir: PathBuf,

    /// Output database path (default: <dir>/.indexnudge.db)
    #[arg(short, long)]
    output: Option<PathBuf>,

    /// Incremental refresh: only re-index files newer than the DB
    #[arg(long)]
    freshen: bool,
}

fn main() -> Result<()> {
    let cli = Cli::parse();
    let dir = cli.dir.canonicalize().context("cannot resolve directory")?;
    let db_path = cli.output.unwrap_or_else(|| dir.join(".indexnudge.db"));

    if cli.freshen {
        freshen(&dir, &db_path)
    } else {
        full_index(&dir, &db_path)
    }
}

fn full_index(dir: &PathBuf, db_path: &PathBuf) -> Result<()> {
    // Remove old DB for full re-index
    if db_path.exists() {
        fs::remove_file(db_path)?;
    }

    let conn = Connection::open(db_path)?;
    create_schema(&conn)?;

    let files = collect_rs_files(dir);
    eprintln!("Indexing {} files...", files.len());

    for file in &files {
        let rel = file.strip_prefix(dir).unwrap_or(file);
        let rel_str = rel.to_string_lossy();
        index_file(&conn, &rel_str, file)?;
    }

    finalize(&conn)?;
    eprintln!("Done. Database: {}", db_path.display());
    Ok(())
}

fn freshen(dir: &PathBuf, db_path: &PathBuf) -> Result<()> {
    if !db_path.exists() {
        eprintln!("No existing database, doing full index.");
        return full_index(dir, db_path);
    }

    let conn = Connection::open(db_path)?;
    conn.execute_batch("PRAGMA journal_mode=WAL; PRAGMA synchronous=OFF;")?;

    // Collect current files on disk
    let disk_files = collect_rs_files(dir);
    let disk_set: std::collections::HashSet<String> = disk_files
        .iter()
        .map(|f| {
            f.strip_prefix(dir)
                .unwrap_or(f)
                .to_string_lossy()
                .to_string()
        })
        .collect();

    // Get indexed files and their mtimes from DB
    let mut stmt = conn.prepare("SELECT file, indexed_mtime FROM files")?;
    let db_files: Vec<(String, f64)> = stmt
        .query_map([], |row| Ok((row.get(0)?, row.get(1)?)))?
        .filter_map(|r| r.ok())
        .collect();
    let db_set: std::collections::HashSet<String> =
        db_files.iter().map(|(f, _)| f.clone()).collect();

    let mut reindexed = 0u64;
    let mut removed = 0u64;

    // Delete files that no longer exist on disk
    for (db_file, _) in &db_files {
        if !disk_set.contains(db_file.as_str()) {
            delete_file_rows(&conn, db_file)?;
            removed += 1;
        }
    }

    // Build a map of db file → indexed_mtime for quick lookup
    let db_mtimes: std::collections::HashMap<&str, f64> =
        db_files.iter().map(|(f, m)| (f.as_str(), *m)).collect();

    // Index new or stale files
    for file in &disk_files {
        let rel = file
            .strip_prefix(dir)
            .unwrap_or(file)
            .to_string_lossy()
            .to_string();
        let disk_mtime = file_mtime(file);

        let needs_index = if let Some(&db_mtime) = db_mtimes.get(rel.as_str()) {
            disk_mtime > db_mtime
        } else {
            true // new file
        };

        if needs_index {
            if db_set.contains(&rel) {
                delete_file_rows(&conn, &rel)?;
            }
            index_file(&conn, &rel, file)?;
            reindexed += 1;
        }
    }

    if reindexed > 0 || removed > 0 {
        // Rebuild directories and FTS
        conn.execute("DELETE FROM directories", [])?;
        conn.execute("DELETE FROM lines_fts", [])?;
        populate_directories(&conn)?;
        conn.execute_batch("INSERT INTO lines_fts(file, line_num, content, symbol) SELECT file, line_num, content, symbol FROM lines;")?;
        eprintln!("Freshened: {} reindexed, {} removed.", reindexed, removed);
    } else {
        eprintln!("All files up to date.");
    }

    Ok(())
}

/// Collect all .rs files under dir, skipping target/
fn collect_rs_files(dir: &PathBuf) -> Vec<PathBuf> {
    WalkDir::new(dir)
        .into_iter()
        .filter_map(|e| e.ok())
        .filter(|e| e.path().extension().is_some_and(|ext| ext == "rs"))
        .filter(|e| !e.path().components().any(|c| c.as_os_str() == "target"))
        .map(|e| e.into_path())
        .collect()
}

/// Index a single file: parse, extract, insert into all tables
fn index_file(conn: &Connection, rel_str: &str, abs_path: &PathBuf) -> Result<()> {
    let source = match fs::read_to_string(abs_path) {
        Ok(s) => s,
        Err(e) => {
            eprintln!("skip {}: {}", rel_str, e);
            return Ok(());
        }
    };

    let syntax = match syn::parse_file(&source) {
        Ok(s) => s,
        Err(e) => {
            eprintln!("parse error in {}: {}", rel_str, e);
            return Ok(());
        }
    };

    index_lines(conn, rel_str, &source, &syntax)?;

    let mut visitor = Extractor::new(rel_str.to_string(), &source);
    visitor.visit_file(&syntax);

    insert_symbols(conn, &visitor.symbols)?;
    insert_calls(conn, &visitor.calls)?;
    insert_imports(conn, &visitor.imports)?;
    insert_defs(conn, &visitor.defs)?;

    let line_count = source.lines().count() as i64;
    let mtime = file_mtime(abs_path);
    conn.execute(
        "INSERT INTO files (file, lines, language, indexed_mtime) VALUES (?1, ?2, ?3, ?4)",
        rusqlite::params![rel_str, line_count, "rust", mtime],
    )?;
    Ok(())
}

/// Delete all rows for a file from every table
fn delete_file_rows(conn: &Connection, file: &str) -> Result<()> {
    conn.execute("DELETE FROM lines WHERE file = ?1", rusqlite::params![file])?;
    conn.execute(
        "DELETE FROM symbols WHERE file = ?1",
        rusqlite::params![file],
    )?;
    conn.execute(
        "DELETE FROM calls WHERE caller_file = ?1",
        rusqlite::params![file],
    )?;
    conn.execute(
        "DELETE FROM imports WHERE file = ?1",
        rusqlite::params![file],
    )?;
    conn.execute("DELETE FROM defs WHERE file = ?1", rusqlite::params![file])?;
    conn.execute("DELETE FROM files WHERE file = ?1", rusqlite::params![file])?;
    Ok(())
}

/// Get file mtime as seconds since epoch (f64)
fn file_mtime(path: &PathBuf) -> f64 {
    fs::metadata(path)
        .and_then(|m| m.modified())
        .map(|t| {
            t.duration_since(std::time::UNIX_EPOCH)
                .unwrap_or_default()
                .as_secs_f64()
        })
        .unwrap_or(0.0)
}

/// Rebuild directories table and populate FTS (used after full index)
fn finalize(conn: &Connection) -> Result<()> {
    populate_directories(conn)?;
    conn.execute_batch("INSERT INTO lines_fts(file, line_num, content, symbol) SELECT file, line_num, content, symbol FROM lines;")?;
    Ok(())
}

fn create_schema(conn: &Connection) -> Result<()> {
    conn.execute_batch(
        "
        PRAGMA journal_mode=WAL;
        PRAGMA synchronous=OFF;

        CREATE TABLE lines (
            file     TEXT,
            line_num INTEGER,
            content  TEXT,
            symbol   TEXT,
            PRIMARY KEY (file, line_num)
        );
        CREATE VIRTUAL TABLE lines_fts USING fts5(file, line_num, content, symbol);

        CREATE TABLE symbols (
            id             INTEGER PRIMARY KEY,
            name           TEXT,
            qualified_name TEXT,
            kind           TEXT,
            file           TEXT,
            start_line     INTEGER,
            end_line       INTEGER,
            parent         TEXT,
            signature      TEXT
        );

        CREATE TABLE calls (
            caller_file   TEXT,
            caller_symbol TEXT,
            callee        TEXT,
            kind          TEXT,
            line_num      INTEGER
        );

        CREATE TABLE defs (
            symbol TEXT,
            file   TEXT,
            line_num INTEGER,
            var    TEXT,
            kind   TEXT
        );

        CREATE TABLE imports (
            file   TEXT,
            module TEXT,
            line_num INTEGER
        );

        CREATE INDEX idx_symbols_name ON symbols(name);
        CREATE INDEX idx_symbols_qualified ON symbols(qualified_name);
        CREATE INDEX idx_symbols_file ON symbols(file);
        CREATE INDEX idx_calls_callee ON calls(callee);
        CREATE INDEX idx_calls_caller ON calls(caller_symbol);
        CREATE INDEX idx_defs_var ON defs(var);
        CREATE INDEX idx_imports_module ON imports(module);

        CREATE TABLE files (
            file           TEXT PRIMARY KEY,
            lines          INTEGER,
            language       TEXT,
            indexed_mtime  REAL
        );

        CREATE TABLE directories (
            path        TEXT PRIMARY KEY,
            file_count  INTEGER,
            total_lines INTEGER
        );
        ",
    )?;
    Ok(())
}

// --- Line indexing ---

fn index_lines(conn: &Connection, file: &str, source: &str, syntax: &syn::File) -> Result<()> {
    // Build a line→symbol map from the parsed file
    let mut line_symbols: Vec<Option<String>> = vec![None; source.lines().count() + 1];
    let mut mapper = LineSymbolMapper {
        lines: &mut line_symbols,
        scope: Vec::new(),
        source,
    };
    mapper.visit_file(syntax);

    let mut stmt = conn
        .prepare("INSERT INTO lines (file, line_num, content, symbol) VALUES (?1, ?2, ?3, ?4)")?;
    for (i, line_content) in source.lines().enumerate() {
        let line_num = (i + 1) as i64;
        let sym = line_symbols.get(i + 1).and_then(|s| s.as_deref());
        stmt.execute(rusqlite::params![file, line_num, line_content, sym])?;
    }
    Ok(())
}

struct LineSymbolMapper<'a> {
    lines: &'a mut Vec<Option<String>>,
    scope: Vec<String>,
    source: &'a str,
}

impl<'a> LineSymbolMapper<'a> {
    fn current_symbol(&self) -> Option<String> {
        self.scope.last().cloned()
    }
}

impl<'a, 'ast> Visit<'ast> for LineSymbolMapper<'a> {
    fn visit_item_fn(&mut self, node: &'ast syn::ItemFn) {
        let name = node.sig.ident.to_string();
        self.scope.push(name);
        let span = node.sig.ident.span();
        let start = span.start().line;
        let end = end_line_of_block(self.source, node.block.brace_token.span.close());
        for l in start..=end {
            if l < self.lines.len() {
                self.lines[l] = self.current_symbol();
            }
        }
        syn::visit::visit_item_fn(self, node);
        self.scope.pop();
    }

    fn visit_impl_item_fn(&mut self, node: &'ast syn::ImplItemFn) {
        let name = node.sig.ident.to_string();
        let qualified = if let Some(parent) = self.scope.last() {
            format!("{}::{}", parent, name)
        } else {
            name.clone()
        };
        self.scope.push(qualified);
        let start = node.sig.ident.span().start().line;
        let end = end_line_of_block(self.source, node.block.brace_token.span.close());
        for l in start..=end {
            if l < self.lines.len() {
                self.lines[l] = self.current_symbol();
            }
        }
        syn::visit::visit_impl_item_fn(self, node);
        self.scope.pop();
    }

    fn visit_item_impl(&mut self, node: &'ast syn::ItemImpl) {
        let name = type_name(&node.self_ty);
        self.scope.push(name);
        syn::visit::visit_item_impl(self, node);
        self.scope.pop();
    }
}

// --- Extraction ---

struct SymbolRecord {
    name: String,
    qualified_name: Option<String>,
    kind: String,
    file: String,
    start_line: usize,
    end_line: usize,
    parent: Option<String>,
    signature: Option<String>,
}

struct CallRecord {
    caller_file: String,
    caller_symbol: Option<String>,
    callee: String,
    kind: String, // "function", "method", "macro"
    line_num: usize,
}

struct DefRecord {
    symbol: Option<String>,
    file: String,
    line_num: usize,
    var: String,
    kind: String, // "let", "assign", "parameter", "mut_ref"
}

struct ImportRecord {
    file: String,
    module: String,
    line_num: usize,
}

struct Extractor {
    file: String,
    source: String,
    symbols: Vec<SymbolRecord>,
    calls: Vec<CallRecord>,
    imports: Vec<ImportRecord>,
    defs: Vec<DefRecord>,
    scope: Vec<String>,     // stack of enclosing symbol names
    impl_type: Vec<String>, // stack of impl type names
}

impl Extractor {
    fn new(file: String, source: &str) -> Self {
        Self {
            file,
            source: source.to_string(),
            symbols: Vec::new(),
            calls: Vec::new(),
            imports: Vec::new(),
            defs: Vec::new(),
            scope: Vec::new(),
            impl_type: Vec::new(),
        }
    }

    fn current_scope(&self) -> Option<String> {
        self.scope.last().cloned()
    }

    fn current_impl(&self) -> Option<&String> {
        self.impl_type.last()
    }

    fn qualified_name(&self, name: &str) -> Option<String> {
        self.current_impl().map(|t| format!("{}::{}", t, name))
    }

    fn sig_text(&self, sig: &syn::Signature) -> String {
        let s = sig.ident.span().start();
        let e = sig.output.span().end();
        extract_span(&self.source, s.line, s.column, e.line, e.column)
    }

    fn add_params_as_defs(&mut self, sig: &syn::Signature) {
        for input in &sig.inputs {
            match input {
                syn::FnArg::Receiver(_) => {}
                syn::FnArg::Typed(pat_type) => {
                    if let Some(name) = pat_name(&pat_type.pat) {
                        self.defs.push(DefRecord {
                            symbol: self.current_scope(),
                            file: self.file.clone(),
                            line_num: pat_type.pat.span().start().line,
                            var: name,
                            kind: "parameter".to_string(),
                        });
                    }
                }
            }
        }
    }
}

use proc_macro2::Span;

fn span_start_line(span: Span) -> usize {
    span.start().line
}

fn span_end_line(span: Span) -> usize {
    span.end().line
}

impl<'ast> Visit<'ast> for Extractor {
    // --- Symbols ---

    fn visit_item_fn(&mut self, node: &'ast syn::ItemFn) {
        let name = node.sig.ident.to_string();
        let start = span_start_line(node.sig.ident.span());
        let end = end_line_of_block(&self.source, node.block.brace_token.span.close());
        self.symbols.push(SymbolRecord {
            name: name.clone(),
            qualified_name: None,
            kind: "function".into(),
            file: self.file.clone(),
            start_line: start,
            end_line: end,
            parent: None,
            signature: Some(self.sig_text(&node.sig)),
        });
        self.scope.push(name);
        self.add_params_as_defs(&node.sig);
        syn::visit::visit_item_fn(self, node);
        self.scope.pop();
    }

    fn visit_item_impl(&mut self, node: &'ast syn::ItemImpl) {
        let ty = type_name(&node.self_ty);
        self.impl_type.push(ty.clone());

        let trait_name = node
            .trait_
            .as_ref()
            .map(|(_, path, _)| path_to_string(path));
        let kind = if trait_name.is_some() {
            "trait_impl"
        } else {
            "impl"
        };

        self.symbols.push(SymbolRecord {
            name: ty.clone(),
            qualified_name: trait_name.as_ref().map(|t| format!("<{} as {}>", ty, t)),
            kind: kind.into(),
            file: self.file.clone(),
            start_line: span_start_line(node.impl_token.span),
            end_line: span_end_line(node.brace_token.span.close()),
            parent: None,
            signature: trait_name.map(|t| format!("impl {} for {}", t, ty)),
        });

        self.scope.push(ty);
        syn::visit::visit_item_impl(self, node);
        self.scope.pop();
        self.impl_type.pop();
    }

    fn visit_impl_item_fn(&mut self, node: &'ast syn::ImplItemFn) {
        let name = node.sig.ident.to_string();
        let qname = self.qualified_name(&name);
        let parent = self.current_impl().cloned();
        let start = span_start_line(node.sig.ident.span());
        let end = end_line_of_block(&self.source, node.block.brace_token.span.close());
        self.symbols.push(SymbolRecord {
            name: name.clone(),
            qualified_name: qname.clone(),
            kind: "method".into(),
            file: self.file.clone(),
            start_line: start,
            end_line: end,
            parent,
            signature: Some(self.sig_text(&node.sig)),
        });
        self.scope.push(qname.unwrap_or(name));
        self.add_params_as_defs(&node.sig);
        syn::visit::visit_impl_item_fn(self, node);
        self.scope.pop();
    }

    fn visit_item_struct(&mut self, node: &'ast syn::ItemStruct) {
        self.symbols.push(SymbolRecord {
            name: node.ident.to_string(),
            qualified_name: None,
            kind: "struct".into(),
            file: self.file.clone(),
            start_line: span_start_line(node.ident.span()),
            end_line: span_end_line(node.ident.span()), // approximate
            parent: None,
            signature: None,
        });
        syn::visit::visit_item_struct(self, node);
    }

    fn visit_item_enum(&mut self, node: &'ast syn::ItemEnum) {
        self.symbols.push(SymbolRecord {
            name: node.ident.to_string(),
            qualified_name: None,
            kind: "enum".into(),
            file: self.file.clone(),
            start_line: span_start_line(node.ident.span()),
            end_line: span_end_line(node.brace_token.span.close()),
            parent: None,
            signature: None,
        });
        syn::visit::visit_item_enum(self, node);
    }

    fn visit_item_trait(&mut self, node: &'ast syn::ItemTrait) {
        let name = node.ident.to_string();
        self.symbols.push(SymbolRecord {
            name: name.clone(),
            qualified_name: None,
            kind: "trait".into(),
            file: self.file.clone(),
            start_line: span_start_line(node.ident.span()),
            end_line: span_end_line(node.brace_token.span.close()),
            parent: None,
            signature: None,
        });
        self.scope.push(name);
        syn::visit::visit_item_trait(self, node);
        self.scope.pop();
    }

    fn visit_trait_item_fn(&mut self, node: &'ast syn::TraitItemFn) {
        let name = node.sig.ident.to_string();
        let qname = self.scope.last().map(|t| format!("{}::{}", t, name));
        self.symbols.push(SymbolRecord {
            name: name.clone(),
            qualified_name: qname,
            kind: "trait_method".into(),
            file: self.file.clone(),
            start_line: span_start_line(node.sig.ident.span()),
            end_line: node
                .default
                .as_ref()
                .map_or(span_end_line(node.sig.ident.span()), |block| {
                    end_line_of_block(&self.source, block.brace_token.span.close())
                }),
            parent: self.scope.last().cloned(),
            signature: Some(self.sig_text(&node.sig)),
        });
    }

    fn visit_item_const(&mut self, node: &'ast syn::ItemConst) {
        self.symbols.push(SymbolRecord {
            name: node.ident.to_string(),
            qualified_name: None,
            kind: "const".into(),
            file: self.file.clone(),
            start_line: span_start_line(node.ident.span()),
            end_line: span_end_line(node.ident.span()),
            parent: None,
            signature: None,
        });
        syn::visit::visit_item_const(self, node);
    }

    fn visit_item_static(&mut self, node: &'ast syn::ItemStatic) {
        self.symbols.push(SymbolRecord {
            name: node.ident.to_string(),
            qualified_name: None,
            kind: "static".into(),
            file: self.file.clone(),
            start_line: span_start_line(node.ident.span()),
            end_line: span_end_line(node.ident.span()),
            parent: None,
            signature: None,
        });
        syn::visit::visit_item_static(self, node);
    }

    fn visit_item_type(&mut self, node: &'ast syn::ItemType) {
        self.symbols.push(SymbolRecord {
            name: node.ident.to_string(),
            qualified_name: None,
            kind: "type".into(),
            file: self.file.clone(),
            start_line: span_start_line(node.ident.span()),
            end_line: span_end_line(node.ident.span()),
            parent: None,
            signature: None,
        });
        syn::visit::visit_item_type(self, node);
    }

    fn visit_item_macro(&mut self, node: &'ast syn::ItemMacro) {
        if let Some(ref ident) = node.ident {
            self.symbols.push(SymbolRecord {
                name: ident.to_string(),
                qualified_name: None,
                kind: "macro".into(),
                file: self.file.clone(),
                start_line: span_start_line(ident.span()),
                end_line: span_end_line(ident.span()),
                parent: None,
                signature: None,
            });
        }
        syn::visit::visit_item_macro(self, node);
    }

    // --- Calls ---

    fn visit_expr_call(&mut self, node: &'ast syn::ExprCall) {
        let callee = expr_callee_name(&node.func);
        if let Some(callee) = callee {
            self.calls.push(CallRecord {
                caller_file: self.file.clone(),
                caller_symbol: self.current_scope(),
                callee,
                kind: "function".into(),
                line_num: span_start_line(node.func.span()),
            });
        }
        syn::visit::visit_expr_call(self, node);
    }

    fn visit_expr_method_call(&mut self, node: &'ast syn::ExprMethodCall) {
        self.calls.push(CallRecord {
            caller_file: self.file.clone(),
            caller_symbol: self.current_scope(),
            callee: node.method.to_string(),
            kind: "method".into(),
            line_num: span_start_line(node.method.span()),
        });
        syn::visit::visit_expr_method_call(self, node);
    }

    fn visit_macro(&mut self, node: &'ast syn::Macro) {
        let name = path_to_string(&node.path);
        self.calls.push(CallRecord {
            caller_file: self.file.clone(),
            caller_symbol: self.current_scope(),
            callee: name,
            kind: "macro".into(),
            line_num: span_start_line(node.path.span()),
        });
        // Don't recurse into macro tokens — syn can't parse them generically
    }

    // --- Imports ---

    fn visit_item_use(&mut self, node: &'ast syn::ItemUse) {
        let modules = flatten_use_tree(&node.tree, &[]);
        for (module, line_num) in modules {
            self.imports.push(ImportRecord {
                file: self.file.clone(),
                module,
                line_num,
            });
        }
    }

    // --- Defs (best effort) ---

    fn visit_local(&mut self, node: &'ast syn::Local) {
        if let Some(name) = pat_name(&node.pat) {
            let kind = if is_pat_mut(&node.pat) {
                "let_mut"
            } else {
                "let"
            };
            self.defs.push(DefRecord {
                symbol: self.current_scope(),
                file: self.file.clone(),
                line_num: span_start_line(node.pat.span()),
                var: name,
                kind: kind.to_string(),
            });
        }
        syn::visit::visit_local(self, node);
    }

    fn visit_expr_assign(&mut self, node: &'ast syn::ExprAssign) {
        if let Some(name) = expr_lhs_name(&node.left) {
            self.defs.push(DefRecord {
                symbol: self.current_scope(),
                file: self.file.clone(),
                line_num: span_start_line(node.left.span()),
                var: name,
                kind: "assign".to_string(),
            });
        }
        syn::visit::visit_expr_assign(self, node);
    }
}

// --- Helpers ---

fn end_line_of_block(_source: &str, close_span: Span) -> usize {
    close_span.end().line
}

fn type_name(ty: &syn::Type) -> String {
    match ty {
        syn::Type::Path(tp) => path_to_string(&tp.path),
        _ => quote::quote!(#ty).to_string(),
    }
}

fn path_to_string(path: &syn::Path) -> String {
    path.segments
        .iter()
        .map(|s| s.ident.to_string())
        .collect::<Vec<_>>()
        .join("::")
}

fn expr_callee_name(expr: &syn::Expr) -> Option<String> {
    match expr {
        syn::Expr::Path(ep) => Some(path_to_string(&ep.path)),
        syn::Expr::Field(ef) => {
            // e.g. self.field — unlikely as a call, but handle it
            Some(ef.member.span().source_text().unwrap_or_default())
        }
        _ => None,
    }
}

fn pat_name(pat: &syn::Pat) -> Option<String> {
    match pat {
        syn::Pat::Ident(pi) => Some(pi.ident.to_string()),
        syn::Pat::Type(pt) => pat_name(&pt.pat),
        _ => None,
    }
}

fn is_pat_mut(pat: &syn::Pat) -> bool {
    match pat {
        syn::Pat::Ident(pi) => pi.mutability.is_some(),
        syn::Pat::Type(pt) => is_pat_mut(&pt.pat),
        _ => false,
    }
}

fn expr_lhs_name(expr: &syn::Expr) -> Option<String> {
    match expr {
        syn::Expr::Path(ep) => Some(path_to_string(&ep.path)),
        syn::Expr::Field(ef) => {
            let base = expr_lhs_name(&ef.base)?;
            let member = match &ef.member {
                syn::Member::Named(ident) => ident.to_string(),
                syn::Member::Unnamed(idx) => idx.index.to_string(),
            };
            Some(format!("{}.{}", base, member))
        }
        syn::Expr::Index(ei) => expr_lhs_name(&ei.expr),
        _ => None,
    }
}

fn flatten_use_tree(tree: &syn::UseTree, prefix: &[String]) -> Vec<(String, usize)> {
    match tree {
        syn::UseTree::Path(p) => {
            let mut new_prefix = prefix.to_vec();
            new_prefix.push(p.ident.to_string());
            flatten_use_tree(&p.tree, &new_prefix)
        }
        syn::UseTree::Name(n) => {
            let mut path = prefix.to_vec();
            path.push(n.ident.to_string());
            vec![(path.join("::"), span_start_line(n.ident.span()))]
        }
        syn::UseTree::Rename(r) => {
            let mut path = prefix.to_vec();
            path.push(r.ident.to_string());
            vec![(path.join("::"), span_start_line(r.ident.span()))]
        }
        syn::UseTree::Glob(g) => {
            let mut path = prefix.to_vec();
            path.push("*".to_string());
            vec![(path.join("::"), span_start_line(g.star_token.span))]
        }
        syn::UseTree::Group(g) => g
            .items
            .iter()
            .flat_map(|t| flatten_use_tree(t, prefix))
            .collect(),
    }
}

fn extract_span(
    source: &str,
    start_line: usize,
    start_col: usize,
    end_line: usize,
    end_col: usize,
) -> String {
    let lines: Vec<&str> = source.lines().collect();
    if start_line == 0 || end_line == 0 || start_line > lines.len() {
        return String::new();
    }
    if start_line == end_line {
        let line = lines[start_line - 1];
        let end = end_col.min(line.len());
        if start_col >= end {
            return line.to_string();
        }
        return line[start_col..end].to_string();
    }
    let mut result = String::new();
    for l in start_line..=end_line.min(lines.len()) {
        if l == start_line {
            result.push_str(&lines[l - 1][start_col..]);
        } else if l == end_line {
            result.push_str(&lines[l - 1][..end_col.min(lines[l - 1].len())]);
        } else {
            result.push_str(lines[l - 1]);
        }
        if l < end_line {
            result.push('\n');
        }
    }
    result
}

fn insert_symbols(conn: &Connection, symbols: &[SymbolRecord]) -> Result<()> {
    let mut stmt = conn.prepare(
        "INSERT INTO symbols (name, qualified_name, kind, file, start_line, end_line, parent, signature)
         VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7, ?8)",
    )?;
    for s in symbols {
        stmt.execute(rusqlite::params![
            s.name,
            s.qualified_name,
            s.kind,
            s.file,
            s.start_line as i64,
            s.end_line as i64,
            s.parent,
            s.signature,
        ])?;
    }
    Ok(())
}

fn insert_calls(conn: &Connection, calls: &[CallRecord]) -> Result<()> {
    let mut stmt = conn.prepare(
        "INSERT INTO calls (caller_file, caller_symbol, callee, kind, line_num)
         VALUES (?1, ?2, ?3, ?4, ?5)",
    )?;
    for c in calls {
        stmt.execute(rusqlite::params![
            c.caller_file,
            c.caller_symbol,
            c.callee,
            c.kind,
            c.line_num as i64,
        ])?;
    }
    Ok(())
}

fn insert_imports(conn: &Connection, imports: &[ImportRecord]) -> Result<()> {
    let mut stmt =
        conn.prepare("INSERT INTO imports (file, module, line_num) VALUES (?1, ?2, ?3)")?;
    for i in imports {
        stmt.execute(rusqlite::params![i.file, i.module, i.line_num as i64])?;
    }
    Ok(())
}

fn insert_defs(conn: &Connection, defs: &[DefRecord]) -> Result<()> {
    let mut stmt = conn.prepare(
        "INSERT INTO defs (symbol, file, line_num, var, kind) VALUES (?1, ?2, ?3, ?4, ?5)",
    )?;
    for d in defs {
        stmt.execute(rusqlite::params![
            d.symbol,
            d.file,
            d.line_num as i64,
            d.var,
            d.kind,
        ])?;
    }
    Ok(())
}

fn populate_directories(conn: &Connection) -> Result<()> {
    use std::collections::HashMap;

    let mut stmt = conn.prepare("SELECT file, lines FROM files")?;
    let rows: Vec<(String, i64)> = stmt
        .query_map([], |row| Ok((row.get(0)?, row.get(1)?)))?
        .filter_map(|r| r.ok())
        .collect();

    let mut dirs: HashMap<String, (i64, i64)> = HashMap::new(); // (file_count, total_lines)
    for (file, lines) in &rows {
        // Collect all ancestor directories
        let path = std::path::Path::new(file);
        let mut ancestors: Vec<String> = Vec::new();
        if let Some(parent) = path.parent() {
            let mut p = parent;
            loop {
                let s = p.to_string_lossy().to_string();
                if s.is_empty() {
                    ancestors.push(".".to_string());
                    break;
                }
                ancestors.push(s);
                match p.parent() {
                    Some(pp) => p = pp,
                    None => break,
                }
            }
        } else {
            ancestors.push(".".to_string());
        }
        for dir in ancestors {
            let entry = dirs.entry(dir).or_insert((0, 0));
            entry.0 += 1;
            entry.1 += lines;
        }
    }

    let mut stmt = conn
        .prepare("INSERT INTO directories (path, file_count, total_lines) VALUES (?1, ?2, ?3)")?;
    for (path, (file_count, total_lines)) in &dirs {
        stmt.execute(rusqlite::params![path, file_count, total_lines])?;
    }
    Ok(())
}

// indexnudge — index a Go codebase into SQLite for agentic querying
// AI-generated code
package main

import (
	"database/sql"
	"flag"
	"fmt"
	"go/ast"
	"go/parser"
	"go/token"
	"go/types"

	"golang.org/x/tools/go/packages"
	"log"
	"os"
	"path/filepath"
	"strings"

	_ "modernc.org/sqlite"
)

func main() {
	dir := flag.String("dir", ".", "directory to index")
	output := flag.String("output", "", "output database path (default: <dir>/.indexnudge.db)")
	freshen := flag.Bool("freshen", false, "incremental refresh: only re-index files newer than the DB")
	flag.Parse()

	absDir, err := filepath.Abs(*dir)
	if err != nil {
		log.Fatal(err)
	}

	dbPath := *output
	if dbPath == "" {
		dbPath = filepath.Join(absDir, ".indexnudge.db")
	}

	if *freshen {
		if err := doFreshen(absDir, dbPath); err != nil {
			log.Fatal(err)
		}
	} else {
		if err := fullIndex(absDir, dbPath); err != nil {
			log.Fatal(err)
		}
	}
}

func fullIndex(dir, dbPath string) error {
	os.Remove(dbPath)
	db, err := sql.Open("sqlite", dbPath)
	if err != nil {
		return err
	}
	defer db.Close()

	if err := createSchema(db); err != nil {
		return err
	}

	fset, pkgFiles, info, err := loadPackages(dir)
	if err != nil {
		return fmt.Errorf("loading packages: %w", err)
	}

	fmt.Fprintf(os.Stderr, "Indexing %d files...\n", len(pkgFiles))

	for absPath, f := range pkgFiles {
		rel, _ := filepath.Rel(dir, absPath)
		if err := indexFile(db, fset, rel, absPath, f, info); err != nil {
			fmt.Fprintf(os.Stderr, "skip %s: %v\n", rel, err)
		}
	}

	if err := finalize(db); err != nil {
		return err
	}
	fmt.Fprintf(os.Stderr, "Done. Database: %s\n", dbPath)
	return nil
}

func doFreshen(dir, dbPath string) error {
	if _, err := os.Stat(dbPath); os.IsNotExist(err) {
		fmt.Fprintln(os.Stderr, "No existing database, doing full index.")
		return fullIndex(dir, dbPath)
	}

	db, err := sql.Open("sqlite", dbPath)
	if err != nil {
		return err
	}
	defer db.Close()

	diskFiles := collectGoFiles(dir)
	diskSet := make(map[string]bool)
	for _, f := range diskFiles {
		rel, _ := filepath.Rel(dir, f)
		diskSet[rel] = true
	}

	rows, err := db.Query("SELECT file, indexed_mtime FROM files")
	if err != nil {
		return err
	}
	type dbFile struct {
		file  string
		mtime float64
	}
	var dbFiles []dbFile
	for rows.Next() {
		var f dbFile
		if err := rows.Scan(&f.file, &f.mtime); err != nil {
			return err
		}
		dbFiles = append(dbFiles, f)
	}
	rows.Close()

	dbMtimes := make(map[string]float64)
	for _, f := range dbFiles {
		dbMtimes[f.file] = f.mtime
	}

	// Check if anything needs updating
	needsWork := false
	for _, f := range dbFiles {
		if !diskSet[f.file] {
			needsWork = true
			break
		}
	}
	if !needsWork {
		for _, absPath := range diskFiles {
			rel, _ := filepath.Rel(dir, absPath)
			diskMtime := fileMtime(absPath)
			dbMtime, exists := dbMtimes[rel]
			if !exists || diskMtime > dbMtime {
				needsWork = true
				break
			}
		}
	}

	if !needsWork {
		fmt.Fprintln(os.Stderr, "All files up to date.")
		return nil
	}

	// Load type info for accurate channel detection
	fset, pkgFiles, info, err := loadPackages(dir)
	if err != nil {
		return fmt.Errorf("loading packages: %w", err)
	}

	var reindexed, removed int

	// Remove deleted files
	for _, f := range dbFiles {
		if !diskSet[f.file] {
			deleteFileRows(db, f.file)
			removed++
		}
	}

	// Index new or stale files
	for _, absPath := range diskFiles {
		rel, _ := filepath.Rel(dir, absPath)
		diskMtime := fileMtime(absPath)
		dbMtime, exists := dbMtimes[rel]
		if !exists || diskMtime > dbMtime {
			if exists {
				deleteFileRows(db, rel)
			}
			if astFile, ok := pkgFiles[absPath]; ok {
				if err := indexFile(db, fset, rel, absPath, astFile, info); err != nil {
					fmt.Fprintf(os.Stderr, "skip %s: %v\n", rel, err)
					continue
				}
			} else {
				// File not in loaded packages (maybe a standalone file)
				localFset := token.NewFileSet()
				src, _ := os.ReadFile(absPath)
				parsed, err := parser.ParseFile(localFset, absPath, src, parser.ParseComments)
				if err != nil {
					fmt.Fprintf(os.Stderr, "skip %s: %v\n", rel, err)
					continue
				}
				if err := indexFile(db, localFset, rel, absPath, parsed, nil); err != nil {
					fmt.Fprintf(os.Stderr, "skip %s: %v\n", rel, err)
					continue
				}
			}
			reindexed++
		}
	}

	if reindexed > 0 || removed > 0 {
		db.Exec("DELETE FROM directories")
		db.Exec("DELETE FROM lines_fts")
		if err := finalize(db); err != nil {
			return err
		}
		fmt.Fprintf(os.Stderr, "Freshened: %d reindexed, %d removed.\n", reindexed, removed)
	}
	return nil
}

// loadPackages uses go/packages to get type information for the whole module.
// Returns the shared FileSet, a map of abs path → parsed AST, and type info.
func loadPackages(dir string) (*token.FileSet, map[string]*ast.File, *types.Info, error) {
	cfg := &packages.Config{
		Mode: packages.NeedFiles | packages.NeedSyntax | packages.NeedTypes |
			packages.NeedTypesInfo | packages.NeedName,
		Dir: dir,
	}
	pkgs, err := packages.Load(cfg, "./...")
	if err != nil {
		return nil, nil, nil, err
	}

	// Merge all type info into one
	info := &types.Info{
		Types: make(map[ast.Expr]types.TypeAndValue),
	}
	result := make(map[string]*ast.File)
	var fset *token.FileSet

	for _, pkg := range pkgs {
		if fset == nil {
			fset = pkg.Fset
		}
		for k, v := range pkg.TypesInfo.Types {
			info.Types[k] = v
		}
		for i, f := range pkg.Syntax {
			if i < len(pkg.GoFiles) {
				result[pkg.GoFiles[i]] = f
			}
		}
	}
	if fset == nil {
		fset = token.NewFileSet()
	}
	return fset, result, info, nil
}

func collectGoFiles(dir string) []string {
	var files []string
	filepath.Walk(dir, func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return nil
		}
		if info.IsDir() {
			base := filepath.Base(path)
			if base == "vendor" || base == ".git" || base == "node_modules" {
				return filepath.SkipDir
			}
			return nil
		}
		if strings.HasSuffix(path, ".go") && !strings.HasSuffix(path, "_test.go") {
			files = append(files, path)
		}
		return nil
	})
	return files
}

func fileMtime(path string) float64 {
	info, err := os.Stat(path)
	if err != nil {
		return 0
	}
	return float64(info.ModTime().UnixMilli()) / 1000.0
}

func createSchema(db *sql.DB) error {
	schema := `
PRAGMA journal_mode=WAL;
PRAGMA synchronous=OFF;

CREATE TABLE lines (
    file     TEXT,
    line_num INTEGER,
    content  TEXT,
    symbol   TEXT,
    PRIMARY KEY (file, line_num)
);
CREATE VIRTUAL TABLE lines_fts USING fts5(file, line_num, content, symbol);

CREATE TABLE symbols (
    id             INTEGER PRIMARY KEY,
    name           TEXT,
    qualified_name TEXT,
    kind           TEXT,
    file           TEXT,
    start_line     INTEGER,
    end_line       INTEGER,
    parent         TEXT,
    signature      TEXT
);

CREATE TABLE calls (
    caller_file   TEXT,
    caller_symbol TEXT,
    callee        TEXT,
    kind          TEXT,
    line_num      INTEGER
);

CREATE TABLE defs (
    symbol TEXT,
    file   TEXT,
    line_num INTEGER,
    var    TEXT,
    kind   TEXT
);

CREATE TABLE imports (
    file   TEXT,
    module TEXT,
    line_num INTEGER
);

CREATE INDEX idx_symbols_name ON symbols(name);
CREATE INDEX idx_symbols_qualified ON symbols(qualified_name);
CREATE INDEX idx_symbols_file ON symbols(file);
CREATE INDEX idx_calls_callee ON calls(callee);
CREATE INDEX idx_calls_caller ON calls(caller_symbol);
CREATE INDEX idx_defs_var ON defs(var);
CREATE INDEX idx_imports_module ON imports(module);

CREATE TABLE files (
    file           TEXT PRIMARY KEY,
    lines          INTEGER,
    language       TEXT,
    indexed_mtime  REAL
);

CREATE TABLE directories (
    path        TEXT PRIMARY KEY,
    file_count  INTEGER,
    total_lines INTEGER
);

CREATE TABLE channels (
    id        INTEGER PRIMARY KEY,
    file      TEXT,
    symbol    TEXT,
    line_num  INTEGER,
    operation TEXT,
    var       TEXT
);
CREATE INDEX idx_channels_var ON channels(var);
CREATE INDEX idx_channels_file ON channels(file);
`
	_, err := db.Exec(schema)
	return err
}

func indexFile(db *sql.DB, fset *token.FileSet, rel, absPath string, f *ast.File, info *types.Info) error {
	src, err := os.ReadFile(absPath)
	if err != nil {
		return err
	}

	lines := strings.Split(string(src), "\n")
	indexLines(db, fset, rel, lines, f)
	extractSymbols(db, fset, rel, f)
	extractCalls(db, fset, rel, f)
	extractImports(db, fset, rel, f)
	extractDefs(db, fset, rel, f)
	extractChannels(db, fset, rel, f, info)

	mtime := fileMtime(absPath)
	db.Exec("INSERT INTO files (file, lines, language, indexed_mtime) VALUES (?, ?, 'go', ?)",
		rel, len(lines), mtime)

	return nil
}

func indexLines(db *sql.DB, fset *token.FileSet, rel string, lines []string, f *ast.File) {
	// Build line→symbol map
	symMap := make(map[int]string)
	ast.Inspect(f, func(n ast.Node) bool {
		switch decl := n.(type) {
		case *ast.FuncDecl:
			name := funcDeclName(decl)
			start := fset.Position(decl.Pos()).Line
			end := fset.Position(decl.End()).Line
			for i := start; i <= end; i++ {
				symMap[i] = name
			}
		}
		return true
	})

	stmt, _ := db.Prepare("INSERT INTO lines (file, line_num, content, symbol) VALUES (?, ?, ?, ?)")
	defer stmt.Close()
	for i, line := range lines {
		lineNum := i + 1
		sym, ok := symMap[lineNum]
		if ok {
			stmt.Exec(rel, lineNum, line, sym)
		} else {
			stmt.Exec(rel, lineNum, line, nil)
		}
	}
}

func extractSymbols(db *sql.DB, fset *token.FileSet, rel string, f *ast.File) {
	stmt, _ := db.Prepare(
		"INSERT INTO symbols (name, qualified_name, kind, file, start_line, end_line, parent, signature) VALUES (?, ?, ?, ?, ?, ?, ?, ?)")
	defer stmt.Close()

	for _, decl := range f.Decls {
		switch d := decl.(type) {
		case *ast.FuncDecl:
			name := d.Name.Name
			kind := "function"
			var parent, qname, sig *string
			if d.Recv != nil {
				kind = "method"
				recvType := recvTypeName(d.Recv)
				if recvType != "" {
					p := recvType
					parent = &p
					q := recvType + "." + name
					qname = &q
				}
			}
			s := funcSignature(d)
			if s != "" {
				sig = &s
			}
			start := fset.Position(d.Pos()).Line
			end := fset.Position(d.End()).Line
			stmt.Exec(name, qname, kind, rel, start, end, parent, sig)

		case *ast.GenDecl:
			for _, spec := range d.Specs {
				switch s := spec.(type) {
				case *ast.TypeSpec:
					kind := "type"
					switch s.Type.(type) {
					case *ast.StructType:
						kind = "struct"
					case *ast.InterfaceType:
						kind = "interface"
					}
					start := fset.Position(s.Pos()).Line
					end := fset.Position(s.End()).Line
					stmt.Exec(s.Name.Name, nil, kind, rel, start, end, nil, nil)
				}
			}
		}
	}
}

func extractCalls(db *sql.DB, fset *token.FileSet, rel string, f *ast.File) {
	stmt, _ := db.Prepare(
		"INSERT INTO calls (caller_file, caller_symbol, callee, kind, line_num) VALUES (?, ?, ?, ?, ?)")
	defer stmt.Close()

	// Walk each function body
	for _, decl := range f.Decls {
		fn, ok := decl.(*ast.FuncDecl)
		if !ok || fn.Body == nil {
			continue
		}
		callerSym := funcDeclName(fn)

		ast.Inspect(fn.Body, func(n ast.Node) bool {
			switch expr := n.(type) {
			case *ast.CallExpr:
				line := fset.Position(expr.Pos()).Line
				switch fun := expr.Fun.(type) {
				case *ast.Ident:
					stmt.Exec(rel, callerSym, fun.Name, "function", line)
				case *ast.SelectorExpr:
					callee := selectorString(fun)
					stmt.Exec(rel, callerSym, callee, "method", line)
				}
			}
			return true
		})
	}
}

func extractImports(db *sql.DB, fset *token.FileSet, rel string, f *ast.File) {
	stmt, _ := db.Prepare("INSERT INTO imports (file, module, line_num) VALUES (?, ?, ?)")
	defer stmt.Close()
	for _, imp := range f.Imports {
		path := strings.Trim(imp.Path.Value, `"`)
		stmt.Exec(rel, path, fset.Position(imp.Pos()).Line)
	}
}

func extractDefs(db *sql.DB, fset *token.FileSet, rel string, f *ast.File) {
	stmt, _ := db.Prepare(
		"INSERT INTO defs (symbol, file, line_num, var, kind) VALUES (?, ?, ?, ?, ?)")
	defer stmt.Close()

	for _, decl := range f.Decls {
		fn, ok := decl.(*ast.FuncDecl)
		if !ok || fn.Body == nil {
			continue
		}
		callerSym := funcDeclName(fn)

		ast.Inspect(fn.Body, func(n ast.Node) bool {
			switch node := n.(type) {
			case *ast.AssignStmt:
				kind := "assign"
				if node.Tok.String() == ":=" {
					kind = "short_var"
				}
				for _, lhs := range node.Lhs {
					if ident, ok := lhs.(*ast.Ident); ok {
						line := fset.Position(ident.Pos()).Line
						stmt.Exec(callerSym, rel, line, ident.Name, kind)
					}
				}
			case *ast.RangeStmt:
				if node.Key != nil {
					if ident, ok := node.Key.(*ast.Ident); ok {
						line := fset.Position(ident.Pos()).Line
						stmt.Exec(callerSym, rel, line, ident.Name, "range_key")
					}
				}
				if node.Value != nil {
					if ident, ok := node.Value.(*ast.Ident); ok {
						line := fset.Position(ident.Pos()).Line
						stmt.Exec(callerSym, rel, line, ident.Name, "range_value")
					}
				}
			}
			return true
		})
	}

	// Package-level vars and consts
	for _, decl := range f.Decls {
		gd, ok := decl.(*ast.GenDecl)
		if !ok {
			continue
		}
		for _, spec := range gd.Specs {
			vs, ok := spec.(*ast.ValueSpec)
			if !ok {
				continue
			}
			kind := "var"
			if gd.Tok.String() == "const" {
				kind = "const"
			}
			for _, name := range vs.Names {
				line := fset.Position(name.Pos()).Line
				stmt.Exec(nil, rel, line, name.Name, kind)
			}
		}
	}
}

func extractChannels(db *sql.DB, fset *token.FileSet, rel string, f *ast.File, info *types.Info) {
	stmt, _ := db.Prepare(
		"INSERT INTO channels (file, symbol, line_num, operation, var) VALUES (?, ?, ?, ?, ?)")
	defer stmt.Close()

	for _, decl := range f.Decls {
		fn, ok := decl.(*ast.FuncDecl)
		if !ok || fn.Body == nil {
			continue
		}
		sym := funcDeclName(fn)

		ast.Inspect(fn.Body, func(n ast.Node) bool {
			switch node := n.(type) {
			case *ast.SendStmt:
				// ch <- v
				line := fset.Position(node.Pos()).Line
				varName := exprName(node.Chan)
				stmt.Exec(rel, sym, line, "send", varName)

			case *ast.UnaryExpr:
				// <-ch
				if node.Op == token.ARROW {
					line := fset.Position(node.Pos()).Line
					varName := exprName(node.X)
					stmt.Exec(rel, sym, line, "recv", varName)
				}

			case *ast.CallExpr:
				if ident, ok := node.Fun.(*ast.Ident); ok {
					switch ident.Name {
					case "make":
						// make(chan T) or make(chan T, n)
						if len(node.Args) > 0 {
							if _, ok := node.Args[0].(*ast.ChanType); ok {
								line := fset.Position(node.Pos()).Line
								stmt.Exec(rel, sym, line, "make", nil)
							}
						}
					case "close":
						if len(node.Args) == 1 {
							line := fset.Position(node.Pos()).Line
							varName := exprName(node.Args[0])
							stmt.Exec(rel, sym, line, "close", varName)
						}
					}
				}

			case *ast.RangeStmt:
				// Only record if ranging over a channel type
				if node.X != nil && info != nil {
					if tv, ok := info.Types[node.X]; ok {
						if _, isChan := tv.Type.Underlying().(*types.Chan); isChan {
							line := fset.Position(node.Pos()).Line
							varName := exprName(node.X)
							stmt.Exec(rel, sym, line, "range", varName)
						}
					}
				}

			case *ast.SelectStmt:
				line := fset.Position(node.Pos()).Line
				stmt.Exec(rel, sym, line, "select", nil)
			}
			return true
		})
	}
}

// Helpers

func funcDeclName(fn *ast.FuncDecl) string {
	if fn.Recv != nil {
		recvType := recvTypeName(fn.Recv)
		if recvType != "" {
			return recvType + "." + fn.Name.Name
		}
	}
	return fn.Name.Name
}

func recvTypeName(recv *ast.FieldList) string {
	if recv == nil || len(recv.List) == 0 {
		return ""
	}
	t := recv.List[0].Type
	// Strip pointer
	if star, ok := t.(*ast.StarExpr); ok {
		t = star.X
	}
	if ident, ok := t.(*ast.Ident); ok {
		return ident.Name
	}
	return ""
}

func funcSignature(fn *ast.FuncDecl) string {
	var b strings.Builder
	b.WriteString("func ")
	if fn.Recv != nil {
		recvType := recvTypeName(fn.Recv)
		if recvType != "" {
			b.WriteString("(" + recvType + ") ")
		}
	}
	b.WriteString(fn.Name.Name + "(")
	if fn.Type.Params != nil {
		var params []string
		for _, p := range fn.Type.Params.List {
			typeName := exprName(p.Type)
			for _, name := range p.Names {
				params = append(params, name.Name+" "+typeName)
			}
			if len(p.Names) == 0 {
				params = append(params, typeName)
			}
		}
		b.WriteString(strings.Join(params, ", "))
	}
	b.WriteString(")")
	if fn.Type.Results != nil && len(fn.Type.Results.List) > 0 {
		var results []string
		for _, r := range fn.Type.Results.List {
			results = append(results, exprName(r.Type))
		}
		if len(results) == 1 {
			b.WriteString(" " + results[0])
		} else {
			b.WriteString(" (" + strings.Join(results, ", ") + ")")
		}
	}
	return b.String()
}

func selectorString(sel *ast.SelectorExpr) string {
	if ident, ok := sel.X.(*ast.Ident); ok {
		return ident.Name + "." + sel.Sel.Name
	}
	return sel.Sel.Name
}

func exprName(expr ast.Expr) string {
	switch e := expr.(type) {
	case *ast.Ident:
		return e.Name
	case *ast.SelectorExpr:
		return selectorString(e)
	case *ast.StarExpr:
		return "*" + exprName(e.X)
	case *ast.ArrayType:
		return "[]" + exprName(e.Elt)
	case *ast.MapType:
		return "map[" + exprName(e.Key) + "]" + exprName(e.Value)
	case *ast.ChanType:
		return "chan " + exprName(e.Value)
	case *ast.FuncType:
		return "func"
	case *ast.InterfaceType:
		return "interface{}"
	case *ast.Ellipsis:
		if e.Elt != nil {
			return "..." + exprName(e.Elt)
		}
		return "..."
	}
	return ""
}

func deleteFileRows(db *sql.DB, file string) {
	for _, table := range []string{"lines", "symbols", "calls", "imports", "defs", "files", "channels"} {
		col := "file"
		if table == "calls" {
			col = "caller_file"
		}
		db.Exec(fmt.Sprintf("DELETE FROM %s WHERE %s = ?", table, col), file)
	}
}

func finalize(db *sql.DB) error {
	populateDirectories(db)
	_, err := db.Exec("INSERT INTO lines_fts(file, line_num, content, symbol) SELECT file, line_num, content, symbol FROM lines")
	return err
}

func populateDirectories(db *sql.DB) {
	rows, err := db.Query("SELECT file, lines FROM files")
	if err != nil {
		return
	}
	defer rows.Close()

	dirs := make(map[string][2]int64) // [file_count, total_lines]
	for rows.Next() {
		var file string
		var lines int64
		if err := rows.Scan(&file, &lines); err != nil {
			continue
		}
		dir := filepath.Dir(file)
		for {
			if dir == "" {
				dir = "."
			}
			entry := dirs[dir]
			entry[0]++
			entry[1] += lines
			dirs[dir] = entry
			if dir == "." {
				break
			}
			dir = filepath.Dir(dir)
		}
	}

	stmt, _ := db.Prepare("INSERT INTO directories (path, file_count, total_lines) VALUES (?, ?, ?)")
	defer stmt.Close()
	for path, counts := range dirs {
		stmt.Exec(path, counts[0], counts[1])
	}
}

func indexSample(t *testing.T, db *sql.DB, src, filename string) {
	t.Helper()
	dir := t.TempDir()
	// loadPackages needs a Go module
	if err := os.WriteFile(filepath.Join(dir, "go.mod"), []byte("module testmod\n\ngo 1.21\n"), 0644); err != nil {
		t.Fatal(err)
	}
	path := filepath.Join(dir, filename)
	if err := os.WriteFile(path, []byte(src), 0644); err != nil {
		t.Fatal(err)
	}
	fset, pkgFiles, info, err := loadPackages(dir)
	if err != nil {
		t.Fatalf("loadPackages: %v", err)
	}
	f, ok := pkgFiles[path]
	if !ok {
		// Try matching by filename suffix — abs paths may differ
		for k, v := range pkgFiles {
			if filepath.Base(k) == filename {
				f = v
				ok = true
				break
			}
		}
		if !ok {
			t.Fatalf("file %s not found in loaded packages (have %v)", filename, pkgFileKeys(pkgFiles))
		}
	}
	if err := indexFile(db, fset, filename, path, f, info); err != nil {
		t.Fatal(err)
	}
}

func queryStrings(t *testing.T, db *sql.DB, query string) []string {
	t.Helper()
	rows, err := db.Query(query)
	if err != nil {
		t.Fatal(err)
	}
	defer rows.Close()
	var result []string
	for rows.Next() {
		var s string
		if err := rows.Scan(&s); err != nil {
			t.Fatal(err)
		}
		result = append(result, s)
	}
	return result
}

func queryInt(t *testing.T, db *sql.DB, query string) int64 {
	t.Helper()
	var n int64
	if err := db.QueryRow(query).Scan(&n); err != nil {
		t.Fatal(err)
	}
	return n
}

func contains(ss []string, target string) bool {

		}
	}
	return false
}

	return keys
}
